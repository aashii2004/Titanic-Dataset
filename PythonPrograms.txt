#Load the tissue_gene_expression dataset. Run a k-means clustering on the data with K=7.
#Make a table comparing the identified clusters to the actual tissue types. Run the algorithm
#several times to see how the answer changes.
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import adjusted_rand_score
import pandas as pd
df=pd.read_csv("https://raw.githubusercontent.com/aashii2004/Titanic-Dataset/refs/heads/main/tissue_gene_expression.csv")
X=df.drop(["rownames","y"],axis=1)
X=StandardScaler().fit_transform(X)
model=KMeans(n_clusters=7,random_state=0,n_init=10)
clusters=model.fit_predict(X)
print("Clusters Vs Actual Tissue Type : ")
table=pd.crosstab(pd.Series(clusters,name="Cluster"),df["y"])
print(table)
for seed in range(5):
    model=KMeans(n_clusters=7,random_state=seed,n_init=10)
    clusters=model.fit_predict(X)
    score=adjusted_rand_score(df["y"],clusters)
    print("\n Run ",seed+1)
    print("Adjusted Rand Score : ",round(score,1))
    print(pd.crosstab(clusters,df["y"]))

from collections import Counter
from itertools import combinations
import pandas as pd
import math

df = pd.read_csv("https://raw.githubusercontent.com/aashii2004/Titanic-Dataset/refs/heads/main/grocery.csv")

transactions = [row.dropna().astype(str).str.strip().tolist() for _, row in df.iterrows()]

min_support = 0.001
min_confidence = 0.8
min_count = math.ceil(min_support * len(transactions))

counts = Counter()
for t in transactions:
    counts.update(set(t))

frequent = {item: count for item, count in counts.items() if count >= min_count}

order = sorted(frequent, key=lambda x: (-frequent[x], x))
rank = {item: i for i, item in enumerate(order)}

ordered_transactions = []
for t in transactions:
    t = [item for item in t if item in frequent]
    t.sort(key=lambda x: rank[x])
    ordered_transactions.append(t)

class Node:
    def __init__(self, item, count=0):
        self.item = item
        self.count = count
        self.children = {}

root = Node("ROOT", 0)

def insert(transaction):
    current = root
    for item in transaction:
        if item not in current.children:
            current.children[item] = Node(item)
        current.children[item].count += 1
        current = current.children[item]

def display(node, level=0):
    print("  " * level + str(node.item) + ":" + str(node.count))
    for child in node.children.values():
        display(child, level + 1)

print("Item Frequencies:")
for item in order:
    print(item, ":", frequent[item])

print("\nGlobal Order:")
print(" => ".join(order))

print("\nFP-Tree Evolution:")
for i, transaction in enumerate(ordered_transactions, 1):
    insert(transaction)
    print(f"\nAfter T{i}: {' -> '.join(transaction)}")
    display(root)

def support_count(items):
    return sum(set(items).issubset(set(t)) for t in transactions)

print("\nAll Association Rules:")
rules_found = False
items = list(frequent.keys())

for size in range(2, len(items) + 1):
    for itemset in combinations(items, size):
        count_all = support_count(itemset)

        if count_all < min_count:
            continue

        for r in range(1, size):
            for antecedent in combinations(itemset, r):
                consequent = tuple(
                    item for item in itemset if item not in antecedent
                )

                count_a = support_count(antecedent)

                if count_a == 0:
                    continue

                confidence = count_all / count_a

                if confidence >= min_confidence:
                    support = count_all / len(transactions)
                    print(
                        f"{set(antecedent)} -> {set(consequent)} "
                        f"| Support: {support:.4f} "
                        f"| Confidence: {confidence:.2f}"
                    )
                    rules_found = True

if not rules_found:
    print("No rules found with confidence >= 0.8")

#Use the Apriori algorithm on the grocery dataset with minimum support to 0.001 and minimum confidence of 0.8 #indicate the top 5 association rules that are generated and
#highlight the strong ones, sort them by confidence.
import pandas as pd
from mlxtend.preprocessing import TransactionEncoder
from mlxtend.frequent_patterns import apriori, association_rules

df = pd.read_csv(
    "https://raw.githubusercontent.com/aashii2004/Titanic-Dataset/refs/heads/main/grocery.csv"
)

transactions = [
    row.dropna().astype(str).str.strip().tolist()
    for _, row in df.iterrows()
]

te = TransactionEncoder()
data = te.fit(transactions).transform(transactions)
basket = pd.DataFrame(data, columns=te.columns_)

frequent_itemsets = apriori(
    basket, min_support=0.001, use_colnames=True
)

rules = association_rules(
    frequent_itemsets,
    metric="confidence",
    min_threshold=0.8
)

rules = rules.sort_values(
    by="confidence", ascending=False
)

rules["Strength"] = rules["confidence"].apply(
    lambda x: "STRONG" if x >= 0.8 else "Not Strong"
)

rules["antecedents"] = rules["antecedents"].apply(
    lambda x: ", ".join(sorted(x))
)
rules["consequents"] = rules["consequents"].apply(
    lambda x: ", ".join(sorted(x))
)

print("TOP 5 ASSOCIATION RULES")
print(
    rules[
        ["antecedents", "consequents", "support",
         "confidence", "lift", "Strength"]
    ].head(5).to_string(index=False)
)

print("\nSTRONG RULES")
strong_rules = rules[rules["confidence"] >= 0.8]

if not strong_rules.empty:
    print(
        strong_rules[
            ["antecedents", "consequents", "support",
             "confidence", "lift", "Strength"]
        ].head(5).to_string(index=False)
    )
else:
    print("No strong rules found.")
