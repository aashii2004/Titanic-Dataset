#service.php

<?php
header("Content-Type: application/json");

$a = $_GET['a'];
$b = $_GET['b'];
$op = $_GET['op'];

if ($op == "add")
    $result = $a + $b;
elseif ($op == "sub")
    $result = $a - $b;
elseif ($op == "mul")
    $result = $a * $b;
elseif ($op == "div") {
    if ($b != 0)
        $result = $a / $b;
    else
        $result = "Cannot divide by zero";
}
else
    $result = "Invalid operation";

echo json_encode(["result" => $result]);
?>

<body ng-controller="myCtrl">
Enter Name : <input type="text" ng-model="name"><br>
Enter Roll No : <input type="text" ng-model="rollNo"></br>
Enter Class : <input type="text" ng-model="class"><br>
<button ng-click="addStudent()">Add Student</button><br>
<h2>Students List</h2>
<table border="1" cellpadding=8">
<th>Name</th><th>RollNo</th><th>Class</th><th>Action</th>
<tr ng-repeat="s in students"><td>{{s.name}}</td><td>{{s.rollNo}}</td><td>{{s.class}}</td><td><button ng-click="deleteStudent($index)">Delete</button></td></tr>
</table>
<script>
var app=angular.module("myApp",[]);
app.controller("myCtrl",function($scope){
$scope.students=[];
$scope.addStudent=function(){
$scope.students.push(
{
name : $scope.name,
rollNo : $scope.rollNo,
class : $scope.class
});
$scope.name="";
$scope.rollNo="";
$scope.class="";
};
$scope.deleteStudent=function(index)
{
$scope.students.splice(index,1);
};
});
</script>
</body>
</html>

<script>
function validateForm()
{
var pid=document.getElementById("pid").value.trim();
var pname=document.getElementById("pname").value.trim();
var address=document.getElementById("address").value.trim();
var city=document.getElementById("city").value.trim();
var cNum=document.getElementById("contact").value.trim();
var dob=document.getElementById("dob").value;
if(pid=="" || pname=="" || address=="" || city=="" || cNum=="" || dob=="")
{
alert("All fields are required.");
return false;
}
if(isNaN(pid))
{
alert("Patient Id must be numeric.");
return false;
}
if(!/^[A-Za-z]+$/.test(pname))
{
alert("Name should contain only letters and spaces.");
return false;
}
if(!/^[A-Za-z]+$/.test(city))
{
alert("City should contain only letters and spaces.");
return false;
}
if(!/^[0-9]{10}$/.test(cNum))
{
alert("Contact Number should contain exactly 10 digits.");
return false;
}
var selectedDate=new Date(dob+"T00:00:00");
var today=new Date();
today.setHours(0,0,0,0);
var hundredYearsAgo=new Date(today);
hundredYearsAgo.setFullYear(today.getFullYear()-100);
if(selectedDate.getTime() > today.getTime())
{
alert("Date cannot be a future date.");
return false;
}
if(selectedDate.getTime() < hundredYearsAgo.getTime())
{
alert("Age cannot be more than 100 years");
return false;
}
alert("Form Submitted Successfully");
return true;
}
window.onload=function(){
var today=new Date();
var hundredYearsAgo=new Date(today);
hundredYearsAgo.setFullYear(today.getFullYear()-100);
document.getElementById("dob").max=today.toISOString().split("T")[0];
document.getElementById("dob").min=hundredYearsAgo.toISOString().split("T")[0];
}
</script>
</head>
